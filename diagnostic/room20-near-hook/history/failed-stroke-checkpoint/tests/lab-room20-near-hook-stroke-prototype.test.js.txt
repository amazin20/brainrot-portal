import test from 'node:test';
import assert from 'node:assert/strict';
import {createHookFixture} from './helpers/room20-hook-prototype-fixture.js';

test('DIAGNOSTIC actual finite-stroke drive uses the same hand impulse under bearing load and after original gravity relief',()=>{
 const f=createHookFixture({strokeDrive:true});try{f.step(240);assert.ok(f.cell.diagnostics().bearingForce>10);const original=f.cargo,shape=f.cargo.shapes[0],driver=f.cell.driverBody;assert.equal(driver.mass,2);assert.equal(f.cell.hand(),true);let e=f.cell.diagnostics().lastHand;assert.equal(e.linearMomentumAfter-e.linearMomentumBefore,5);f.step(240);assert.ok(Math.abs(f.cell.opening)<.004);assert.ok(f.trace.slice(240).every(r=>r.hookAngle<.4&&r.opening<.4));f.releaseTray();f.step(600);let d=f.cell.diagnostics();assert.equal(d.bearingForce,0);assert.ok(d.captureForce>20&&d.tension>50);assert.equal(f.cell.hand(),true);e=f.cell.diagnostics().lastHand;assert.ok(Math.abs(e.linearMomentumAfter-e.linearMomentumBefore-5)<1e-12);f.step(480);d=f.cell.diagnostics();assert.ok(d.opening>2.39&&d.hookAngle>2.7);assert.equal(d.drive.registryValid,true);f.cell.cutCable();f.step(960);assert.ok(Math.abs(f.cell.opening+.05)<1e-6);assert.equal(f.cargo,original);assert.equal(f.cargo.shapes[0],shape);assert.equal(f.cell.originalCargoBody,original);assert.ok(f.raw.maxBearingDepth<.035);assert.ok(f.raw.maxHingeDrift<.005);console.info('ROOM20_HOOK_STROKE_SINGLE',JSON.stringify({ordinaryAccepted:false,prototypeAccepted:false,handMode:d.handMode,inputImpulse:5,opened:d,returned:f.cell.diagnostics(),maxBearingDepth:f.raw.maxBearingDepth,maxHingeDrift:f.raw.maxHingeDrift}));}finally{f.dispose();}
});

test('DIAGNOSTIC finite-stroke owned drive keeps same IDs across reset and removes both callback sets before any owned body',()=>{
 const f=createHookFixture({strokeDrive:true});try{const {cell,world,game}=f;f.step(240);const ids=cell.bodies.map(b=>b.id),owned=[...cell.bodies],unowned=world.bodies.filter(b=>!owned.includes(b)),cp=f.cargo.position.clone(),cq=f.cargo.quaternion.clone(),foreign={update(){}};world.subsystems.push(foreign);const callbacks=[...world.subsystems];cell.reset();cell.reset();assert.deepEqual(cell.bodies.map(b=>b.id),ids);assert.deepEqual(world.subsystems,callbacks);assert.equal(f.cargo.position.distanceTo(cp),0);assert.deepEqual(f.cargo.quaternion.toArray(),cq.toArray());assert.equal(cell.diagnostics().drive.registryValid,true);const remove=world.removeBody.bind(world);world.removeBody=b=>{if(owned.includes(b)){assert.deepEqual(world.subsystems,[foreign]);assert.ok([...game.physics.bodyBoxes.values()].every(e=>e.body!==b));assert.ok([...game.physics.cargoSweepBoxes.values()].every(e=>e.body!==b));assert.ok(world.constraints.every(c=>c.bodyA!==b&&c.bodyB!==b));}return remove(b);};cell.dispose();cell.dispose();assert.deepEqual(world.bodies,unowned);assert.equal(game.physics.bodyBoxes.size,0);assert.equal(game.physics.cargoSweepBoxes.size,0);assert.equal(cell.hand(),false);}finally{f.dispose();}
});

test('DIAGNOSTIC reached 30 Hz input must not mechanically remove loaded hook without original traction',()=>{
 const f=createHookFixture({strokeDrive:true});try{f.cell.cutCable();f.step(240,{record:false});const initial=f.cell.diagnostics();assert.ok(initial.bearingForce>15&&initial.bearingContacts>0&&initial.originalFree);assert.equal(initial.tension,0);let requests=0,maxAngle=initial.hookAngle,maxOpening=initial.opening;for(let i=0;i<3600;i++){if(i%4===0){assert.equal(f.cell.hand(),true);requests++;}f.step(1,{record:false});const d=f.cell.diagnostics();assert.equal(d.tension,0);maxAngle=Math.max(maxAngle,d.hookAngle);maxOpening=Math.max(maxOpening,d.opening);}assert.equal(requests,900);assert.equal(f.cell.diagnostics().handCount,900);console.info('ROOM20_HOOK_STROKE_REACHED_SPAM',JSON.stringify({ordinaryAccepted:false,prototypeAccepted:false,requests,initial,final:f.cell.diagnostics(),maxAngle,maxOpening}));assert.ok(maxAngle<.7&&maxOpening<.4,'OPEN BLOCKER: finite stroke still transfers enough repeated physical hand work to bypass original-cargo unloading');}finally{f.dispose();}
});
